    function shuffle(arr) {
      const a = [...arr];
      for (let i = a.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [a[i], a[j]] = [a[j], a[i]];
      }
      return a;
    }

    function loadStore() {
      try {
        const d = JSON.parse(localStorage.getItem(STORE_KEY) || '{}');
        return {
          totalAttempts: Number(d.totalAttempts) || 0,
          solvedIds: Array.isArray(d.solvedIds) ? d.solvedIds : [],
          errors: Array.isArray(d.errors) ? d.errors : [],
          bestExam: d.bestExam || null,
          bestBlock: d.bestBlock || {}
        };
      } catch {
        return { totalAttempts: 0, solvedIds: [], errors: [], bestExam: null, bestBlock: {} };
      }
    }

    function saveStore(d) {
      try { localStorage.setItem(STORE_KEY, JSON.stringify(d)); } catch (e) {}
    }

    function trackAnswer(qid, isOk, removeOnOk) {
      const st = loadStore();
      st.totalAttempts += 1;
      if (isOk) {
        if (!st.solvedIds.includes(qid)) st.solvedIds.push(qid);
        if (removeOnOk) st.errors = st.errors.filter(x => x !== qid);
      } else {
        if (!st.errors.includes(qid)) st.errors.push(qid);
      }
      saveStore(st);
    }

    function saveBest(key, ok, tot, pct) {
      const st = loadStore();
      if (key === 'exam40') {
        if (!st.bestExam || ok > st.bestExam.ok) st.bestExam = { ok, tot, pct };
      } else if (key) {
        const prev = st.bestBlock[key];
        if (!prev || pct > prev.pct) st.bestBlock[key] = { ok, tot, pct };
      }
      saveStore(st);
    }

    function renderCrib() {
      const qStr = (document.getElementById('crib-search').value || '').trim().toLowerCase();
      const numMatch = qStr.startsWith('#') ? qStr.slice(1) : qStr;

      const filtered = QUESTIONS.filter(item => {
        if (state.cribFilter === 'visual' && !item.hasVisual) return false;
        if (state.cribFilter === 'code' && !item.hasCode) return false;
        if (!['all', 'visual', 'code'].includes(state.cribFilter)) {
          const tm = TOPICS.find(t => t.id === state.cribFilter);
          if (tm && !tm.tags.includes(item.topic)) return false;
        }
        if (!qStr) return true;
        if (String(item.id) === numMatch) return true;
        const hay = `${item.id} ${item.q} ${item.code} ${item.opts.join(' ')}`.toLowerCase();
        return hay.includes(qStr);
      });

      document.getElementById('crib-count').textContent = `Табылды: ${filtered.length} / ${QUESTIONS.length}`;
      const listEl = document.getElementById('crib-list');
      listEl.innerHTML = '';

      if (!filtered.length) {
        listEl.innerHTML = '
Іздеу бойынша сұрақ табылмады 🔍
';
        return;
      }

      const frag = document.createDocumentFragment();
      filtered.forEach(item => {
        const tm = getTopicMeta(item.topic);
        const div = document.createElement('div');
        div.className = 'crib-card';
        const correctRaw = item.opts[item.ans];
        const correctDisplay = OPT_VISUALS[correctRaw]
          ? `${LETTERS[item.ans]})${OPT_VISUALS[correctRaw]}`
          : `${LETTERS[item.ans]}) ${esc(correctRaw)}`;

        let html = `
          

            #${item.id}
            ${tm.icon} ${esc(tm.name)}
          

          
${esc(item.q)}
`;
        if (item.code) html += `
${esc(item.code)}
`;
        if (VISUALS[item.id]) html += `
${VISUALS[item.id]}
`;
        html += `
✅ ${correctDisplay}
`;
        div.innerHTML = html;
        frag.appendChild(div);
      });
      listEl.appendChild(frag);
    }

    function renderQ() {
      const s = state.session;
      if (s.type !== 'exam') stopTimer();

      const q = s.list[s.idx];
      const tot = s.list.length;
      const ua = s.ans[s.idx];
      const answered = ua !== null;
      const isExam = s.type === 'exam';
      const isStudy = !isExam && state.mode === 'study';
      const reveal = answered && s.instant;

      document.getElementById('hud-pos').textContent = `${s.idx + 1} / ${tot}`;
      document.getElementById('hud-qid').textContent = `#${q.id}`;
      document.getElementById('hud-score').textContent = s.score.toLocaleString('ru-RU');

      const doneCnt = s.ans.filter(a => a !== null).length;
      document.getElementById('hud-prog').style.width = `${(doneCnt / tot) * 100}%`;

      // Flag button state
      const fBtn = document.getElementById('btn-flag');
      fBtn.classList.toggle('flagged', Boolean(s.flags[s.idx]));

      // Streak badge
      const sb = document.getElementById('hud-streak-box');
      if (s.instant && !isStudy && s.streak >= 2) {
        sb.classList.remove('hidden');
        document.getElementById('hud-streak').textContent = s.streak;
      } else {
        sb.classList.add('hidden');
      }

      // Topic & Mode labels
      const tm = getTopicMeta(q.topic);
      document.getElementById('q-topic-badge').textContent = `${tm.icon} ${tm.name}`;
      document.getElementById('q-mode-lbl').textContent = isExam
        ? (s.instant ? 'Емтихан (Бірден тексеру)' : 'Нақты емтихан')
        : (isStudy ? '📚 Жаттау режимі' : '🎮 Ойын режимі');

      // Question text
      document.getElementById('q-text').textContent = q.q;

      // Code block
      const codeEl = document.getElementById('q-code');
      if (q.code) {
        codeEl.classList.remove('hidden');
        codeEl.textContent = q.code;
      } else {
        codeEl.classList.add('hidden');
      }

      // Visual block (Excel/SQL/Truth/SVG)
      const visEl = document.getElementById('q-visual');
      if (VISUALS[q.id]) {
        visEl.classList.remove('hidden');
        visEl.innerHTML = VISUALS[q.id];
      } else {
        visEl.classList.add('hidden');
        visEl.innerHTML = '';
      }

      // 4 Answer Option Tiles
      const grid = document.getElementById('opts-grid');
      grid.innerHTML = '';

      q.opts.forEach((opt, i) => {
        const btn = document.createElement('button');
        btn.type = 'button';
        let cls = `tile t-${i % 4}`;

        if (reveal) {
          btn.disabled = true;
          if (i === q.ans) cls += ' t-ok';
          else if (i === ua) cls += ' t-err';
          else cls += ' t-dim';
        } else if (answered && !s.instant && i === ua) {
          cls += ' t-sel';
        }

        btn.className = cls;
        btn.onclick = () => pickOpt(i);

        let mark = '';
        if (reveal) {
          if (i === q.ans) mark = '✓';
          else if (i === ua) mark = '✕';
        } else if (answered && !s.instant && i === ua) {
          mark = '●';
        }

        btn.innerHTML = `
          ${LETTERS[i]}
          ${renderOptHTML(opt)}
          ${mark}`;
        grid.appendChild(btn);
      });

      // Bottom buttons & feedback
      const fb = document.getElementById('q-feedback');
      const nextB = document.getElementById('btn-next');
      const prevB = document.getElementById('btn-prev');

      nextB.textContent = s.idx === tot - 1 ? 'Аяқтау 🏁' : 'Келесі →';

      if (isStudy || isExam) {
        prevB.disabled = s.idx === 0;
        nextB.disabled = false;
        renderNavGrid();
      } else {
        nextB.disabled = !answered;
      }

      if (!answered) {
        fb.style.color = 'var(--soft)';
        fb.textContent = 'Жауап нұсқасын таңдаңыз (A–D немесе 1–4)';
        if (!isStudy && !isExam) startGameTimer();
      } else if (!s.instant) {
        fb.style.color = '#c4b5fd';
        fb.textContent = `Таңдалды: ${LETTERS[ua]} (соңында тексеріледі — басқасын таңдап өзгертуге болады)`;
      } else if (ua === q.ans) {
        fb.style.color = '#6ee7b7';
        fb.textContent = '✅ Дұрыс жауап!';
      } else if (ua === -1) {
        fb.style.color = '#fde68a';
        fb.textContent = `⏰ Уақыт бітті! Дұрыс жауап: ${LETTERS[q.ans]}`;

        fb.style.color = '#fda4af';

    function startGameTimer() {
      const s = state.session;
      s.timeLeft = GAME_SEC;
      updateTimerUI();
      s.timer = setInterval(() => {
        s.timeLeft--;
        updateTimerUI();

    function pickOpt(i) {
      const s = state.session;
      const q = s.list[s.idx];

      if (!s.instant) {
        SFX.click();
        s.ans[s.idx] = i;
        renderQ();
        return;
      }

      if (s.ans[s.idx] !== null) return;
      if (s.type !== 'exam') stopTimer();

      s.ans[s.idx] = i;
      const isOk = i === q.ans;
      trackAnswer(q.id, isOk, s.type === 'errors');

      if (isOk) {
        s.streak++;
        if (s.streak > s.bestStreak) s.bestStreak = s.streak;
        if (s.type === 'exam' || state.mode === 'game') {
          const spd = s.type === 'exam' ? 150 : Math.round((s.timeLeft / GAME_SEC) * 400);
          const str = Math.min(s.streak - 1, 5) * 80;
          const pts = 1000 + spd + str;
          s.score += pts;
          popScore(pts);
        } else {
          s.score += 100;
        }
        SFX.ok(s.streak);
      } else {
        s.streak = 0;
        SFX.err();
      }
      renderQ();
    }

          : 'Барлық 40 сұраққа жауап берілді! Емтиханды аяқтап, нәтижеңізді көресіз бе?';

    /* =========================================================================
       9. FINISH QUIZ, REVIEW & CANVAS CONFETTI
       ========================================================================= */
    function finishQuiz() {
      stopTimer();
      const s = state.session;
      const elapsed = Math.max(1, Math.round((Date.now() - s.startedAt) / 1000));
      const tot = s.list.length;
      let ok = 0;

      if (!s.instant) {
        let cur = 0;
        s.bestStreak = 0;
        s.score = 0;
        s.list.forEach((q, i) => {
          const good = s.ans[i] === q.ans;
          trackAnswer(q.id, good, false);
          if (good) {
            ok++; cur++;
            if (cur > s.bestStreak) s.bestStreak = cur;
            s.score += 1000 + Math.min(cur - 1, 5) * 80;
          } else {
            cur = 0;
          }
        });
      } else {
        s.list.forEach((q, i) => { if (s.ans[i] === q.ans) ok++; });
      }

      const wrong = tot - ok;
      const pct = tot ? Math.round((ok / tot) * 100) : 0;
      const passed = s.type === 'exam' ? ok >= 32 : pct >= 80;

      saveBest(s.key, ok, tot, pct);

      if (passed) { SFX.win(); runConfetti(); }
      else SFX.click();

      document.getElementById('res-pct').textContent = `${pct}%`;
      document.getElementById('res-ico').textContent = passed ? '🏆' : pct >= 60 ? '🔥' : '💪';
      document.getElementById('res-title').textContent = passed ? 'Керемет нәтиже!' : pct >= 60 ? 'Жақсы нәтиже!' : 'Қателермен жұмыс жасайық!';
      document.getElementById('res-sub').textContent = s.title;
      document.getElementById('res-ok').textContent = `${ok} / ${tot}`;
      document.getElementById('res-pts').textContent = s.score.toLocaleString('ru-RU');
      document.getElementById('res-strk').textContent = `🔥 ${s.bestStreak}`;
      document.getElementById('res-time').textContent = fmtTime(elapsed);

      const bdg = document.getElementById('res-badge');

    function renderReview(filter) {
      document.getElementById('rev-btn-all').classList.toggle('active', filter === 'all');
      document.getElementById('rev-btn-err').classList.toggle('active', filter === 'wrong');

      const s = state.session;
      const list = document.getElementById('rev-list');
      list.innerHTML = '';
      let shown = 0;

      s.list.forEach((q, i) => {
        const ua = s.ans[i];
        const good = ua === q.ans;
        if (filter === 'wrong' && good) return;
        shown++;

        const d = document.createElement('details');
        d.className = `rev-item ${good ? 'rev-ok' : 'rev-err'}`;
        if (!good) d.open = true;

        const uHtml = ua === null ? 'Белгіленбеген' : ua === -1 ? 'Уақыт бітті' : `${LETTERS[ua]}) ${renderOptHTML(q.opts[ua])}`;
        const rHtml = `${LETTERS[q.ans]}) ${renderOptHTML(q.opts[q.ans])}`;

        let body = '';
        if (q.code) body += `
${esc(q.code)}
`;
        if (VISUALS[q.id]) body += `
${VISUALS[q.id]}
`;
        body += `
Сіздің жауабыңыз: ${uHtml}
`;
        if (!good) body += `
Дұрыс жауап: ${rHtml}
`;

        d.innerHTML = `
          
            ${good ? '✅' : '❌'} ${i + 1}. [#${q.id}] ${esc(q.q)}
            ▼
          
          
${body}
`;
        list.appendChild(d);
      });

      if (!shown) {
        list.innerHTML = '
🎉 Қате жоқ! Барлық сұрақ дұрыс!
';
      }
    }

    /* Canvas Confetti */
    let confRaf = null;
    function runConfetti() {
      const c = document.getElementById('confetti-canvas');
      const ctx = c.getContext('2d');
      c.classList.remove('hidden');
      const resize = () => { c.width = window.innerWidth; c.height = window.innerHeight; };
      resize();
      window.addEventListener('resize', resize);

      const colors = ['#7c3aed', '#06b6d4', '#10b981', '#f59e0b', '#f43f5e', '#3b82f6'];
      const pts = Array.from({ length: 130 }, () => ({
        x: Math.random() * c.width,
        y: -20 - Math.random() * c.height * 0.5,
        w: 7 + Math.random() * 6,
        h: 10 + Math.random() * 8,
        vx: (Math.random() - 0.5) * 4.5,
        vy: 2.5 + Math.random() * 4.5,
        rot: Math.random() * Math.PI * 2,
        vr: (Math.random() - 0.5) * 0.22,
        col: colors[Math.floor(Math.random() * colors.length)]
      }));

      const t0 = performance.now();
      if (confRaf) cancelAnimationFrame(confRaf);
      function tick(now) {
        const dt = now - t0;
        ctx.clearRect(0, 0, c.width, c.height);
        ctx.globalAlpha = dt > 3200 ? Math.max(0, 1 - (dt - 3200) / 1200) : 1;
        pts.forEach(p => {
          p.x += p.vx; p.y += p.vy; p.rot += p.vr;
          ctx.save(); ctx.translate(p.x, p.y); ctx.rotate(p.rot);
          ctx.fillStyle = p.col; ctx.fillRect(-p.w / 2, -p.h / 2, p.w, p.h);
          ctx.restore();
        });
        if (dt < 4400) confRaf = requestAnimationFrame(tick);
        else { ctx.clearRect(0, 0, c.width, c.height); c.classList.add('hidden'); window.removeEventListener('resize', resize); }
      }
      confRaf = requestAnimationFrame(tick);
    }

    document.addEventListener('keydown', e => {
      const tag = document.activeElement && document.activeElement.tagName;
      if (tag === 'INPUT' || tag === 'TEXTAREA') return;
      if (e.key === 'Escape') {
        ['modal-exit', 'modal-finish', 'modal-clear-err', 'drawer-map'].forEach(closeModal);
        return;
      }
      const inQuiz = !document.getElementById('screen-quiz').classList.contains('hidden');
      const modalUp = ['modal-exit', 'modal-finish', 'drawer-map'].some(id => !document.getElementById(id).classList.contains('hidden'));
      if (!inQuiz || modalUp) return;

      const k = e.key.toLowerCase();
      const kMap = { '1': 0, '2': 1, '3': 2, '4': 3, 'a': 0, 'b': 1, 'c': 2, 'd': 3 };
      if (k in kMap) {
        e.preventDefault();
        pickOpt(kMap[k]);
      } else if (e.key === 'Enter' || e.key === ' ') {
        const nb = document.getElementById('btn-next');
        if (!nb.disabled) { e.preventDefault(); nextQ(); }
      } else if (e.key === 'ArrowRight') {
        const nb = document.getElementById('btn-next');
        if (!nb.disabled) { e.preventDefault(); nextQ(); }
      } else if (e.key === 'ArrowLeft') {
        const pb = document.getElementById('btn-prev');
        if (!pb.classList.contains('hidden') && !pb.disabled) { e.preventDefault(); prevQ(); }
      }
    });
